# Builds the gpufx MLT module on its own, outside of Kdenlive's build, for
# example to add it to an installed Kdenlive:
#
#   cmake -S gpufx/standalone -B build
#   cmake --build build
#   cmake --install build --prefix <Kdenlive's folder>
#
# MLT is found through its CMake package (Mlt7). An MLT without development
# files, like the one in Kdenlive's Windows standalone build, can be given as
# headers (the src directory of MLT's source) and its library instead:
#
#   -DMLT_INCLUDE_DIR=<MLT source>/src -DMLT_LIBRARY=<Kdenlive>/bin/libmlt-7.dll
#
# MinGW links DLLs directly, no import library is needed.

cmake_minimum_required(VERSION 3.16)
project(gpufx LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

include(GNUInstallDirs)
include(CTest)
set(KDE_INSTALL_LIBDIR "${CMAKE_INSTALL_LIBDIR}")

set(MLT_INCLUDE_DIR "" CACHE PATH "MLT headers (the directory with framework/mlt.h), for an MLT without CMake package")
set(MLT_LIBRARY "" CACHE FILEPATH "MLT library, for an MLT without CMake package")
if(MLT_INCLUDE_DIR AND MLT_LIBRARY)
    set(mlt_include_dirs "${MLT_INCLUDE_DIR}")
    if(NOT EXISTS "${MLT_INCLUDE_DIR}/framework/mlt_export.h")
        # MLT's build generates this header, which is not in its source
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mlt_export/mlt_export.h" [[
/* mlt_export.h as generated by MLT's build, for users of the library */
#ifndef MLT_EXPORT_H
#define MLT_EXPORT_H
#ifdef _WIN32
#define MLT_EXPORT __declspec(dllimport)
#else
#define MLT_EXPORT __attribute__((visibility("default")))
#endif
#define MLT_NO_EXPORT
#define MLT_DEPRECATED __attribute__((__deprecated__))
#define MLT_DEPRECATED_EXPORT MLT_EXPORT MLT_DEPRECATED
#define MLT_DEPRECATED_NO_EXPORT MLT_NO_EXPORT MLT_DEPRECATED
#endif
]])
        list(APPEND mlt_include_dirs "${CMAKE_CURRENT_BINARY_DIR}/mlt_export")
    endif()
    add_library(Mlt7::mlt UNKNOWN IMPORTED)
    set_target_properties(Mlt7::mlt PROPERTIES
        IMPORTED_LOCATION "${MLT_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${mlt_include_dirs}")
else()
    find_package(Mlt7 REQUIRED)
endif()

if(MINGW)
    # Kdenlive's own GCC runtime may be older than the one used here: link it
    # statically, so that the module only needs MLT and system libraries
    option(GPUFX_STATIC_RUNTIME "Link the GCC runtime libraries statically" ON)
    if(GPUFX_STATIC_RUNTIME)
        add_link_options(-static)
    endif()
endif()

add_subdirectory(.. gpufx)
